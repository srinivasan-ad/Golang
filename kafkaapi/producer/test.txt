func Validate(u *UserType) bool {
        return gt(len(u.UserName), 5) && lt(len(u.UserName), 50) && gt(u.Id, 1) && lt(u.Id, 100) && gt(len(u.Email), 10) && lt(len(u.Email), 100) && lt(u.Address.Zip, 9999) && gt(len(u.Address.Country.Name), 3) && lt(len(u.Address.Country.Name), 50) && gt(len(u.Address.Country.Code), 2) && lt(len(u.Address.Country.Code), 10) && gt(len(u.Address.Street), 10) && lt(len(u.Address.Street), 100) && gt(len(u.Address.City), 3) && lt(len(u.Address.City), 50) && gt(len(u.Preferences.Theme), 3) && lt(len(u.Preferences.Theme), 20) && gt(len(u.Preferences.Language), 2) && lt(len(u.Preferences.Language), 10)
}
